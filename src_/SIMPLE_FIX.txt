<script>
(function() {
    'use strict';
    const API = 'http://localhost:8082/api/chat/message';
    const WELCOME = '👋 Xin chào! Tôi là EduGuard Assistant.\n\n📊 Xem điểm số\n📅 Kiểm tra chuyên cần\n⚠️ Theo dõi sinh viên nguy cơ\n📈 Thống kê lớp học';
    
    function getUser() {
        if (window.M?.cfg?.username) {
            const u = window.M.cfg.username;
            const id = window.M.cfg.userid || 0;
            let role = 'GUEST';
            if (u === 'admin') role = 'ADMIN';
            else if (u.startsWith('giangvien') || u.startsWith('teacher') || u.startsWith('gv')) role = 'LECTURER';
            else if (u.match(/^11\d{7}$/) || u.match(/^student\d+$/)) role = 'STUDENT';
            else if (u !== 'guest' && id > 0) role = 'STUDENT';
            return { username: u, userid: id > 0 ? id : 1, role: role, sessionId: 'session-' + Date.now() };
        }
        return { username: 'guest', userid: 1, role: 'GUEST', sessionId: 'session-' + Date.now() };
    }
    
    function addMsg(text, sender) {
        const c = document.getElementById('eduguard-messages');
        if (!c) return;
        const div = document.createElement('div');
        div.className = 'eduguard-msg ' + sender;
        const content = document.createElement('div');
        content.className = 'eduguard-msg-content';
        content.textContent = text;
        div.appendChild(content);
        c.appendChild(div);
        c.scrollTop = c.scrollHeight;
    }
    
    function showTyping() {
        const c = document.getElementById('eduguard-messages');
        if (!c) return;
        const div = document.createElement('div');
        div.className = 'eduguard-msg bot';
        div.id = 'typing';
        const content = document.createElement('div');
        content.className = 'eduguard-msg-content';
        content.innerHTML = '<div class="eduguard-typing"><span></span><span></span><span></span></div>';
        div.appendChild(content);
        c.appendChild(div);
        c.scrollTop = c.scrollHeight;
    }
    
    function hideTyping() {
        const t = document.getElementById('typing');
        if (t) t.remove();
    }
    
    async function send(msg) {
        const user = getUser();
        const btn = document.getElementById('eduguard-send');
        if (!btn) return;
        btn.disabled = true;
        showTyping();
        try {
            const res = await fetch(API, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ message: msg, username: user.username, role: user.role, sessionId: user.sessionId, userId: user.userid })
            });
            hideTyping();
            if (!res.ok) throw new Error('HTTP ' + res.status);
            const data = await res.json();
            const reply = data.data?.reply || data.message || 'Xin lỗi, không thể xử lý.';
            addMsg(reply, 'bot');
        } catch (err) {
            console.error('EduGuard:', err);
            hideTyping();
            addMsg('⚠️ Lỗi kết nối server (port 8082).', 'bot');
        } finally {
            btn.disabled = false;
        }
    }
    
    function handleSend() {
        const input = document.getElementById('eduguard-input');
        if (!input) return;
        const msg = input.value.trim();
        if (msg) {
            addMsg(msg, 'user');
            input.value = '';
            send(msg);
        }
    }
    
    function init() {
        const btn = document.getElementById('eduguard-send');
        const input = document.getElementById('eduguard-input');
        const minimize = document.getElementById('eduguard-minimize');
        
        if (!btn || !input) return;
        
        btn.addEventListener('click', handleSend);
        input.addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                e.preventDefault();
                handleSend();
            }
        });
        
        if (minimize) {
            minimize.addEventListener('click', function() {
                const body = document.getElementById('eduguard-chat-body');
                if (body) {
                    body.classList.toggle('minimized');
                }
            });
        }
        
        addMsg(WELCOME, 'bot');
        console.log('✅ EduGuard: Ready in footer');
    }
    
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        setTimeout(init, 100);
    }
})();
</script>
